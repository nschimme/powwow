cmake_minimum_required(VERSION 3.16)
project(powwow VERSION 2.0.0 LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Create dummy config.h in binary directory for C sources
if(NOT EXISTS "${CMAKE_BINARY_DIR}/config.h")
    file(WRITE "${CMAKE_BINARY_DIR}/config.h" "/* Generated by CMake */\n")
endif()

# Build Options
option(ENABLE_TLS "Enable TLS support via GnuTLS" ON)
option(BUILD_GUI "Build Qt6 GUI Launcher and Terminal" ON)

# Common compiler and feature definitions applied via interface target
add_library(powwow_platform_definitions INTERFACE)
target_include_directories(powwow_platform_definitions INTERFACE
    "${CMAKE_BINARY_DIR}"
    "${CMAKE_SOURCE_DIR}/include"
)
target_compile_definitions(powwow_platform_definitions INTERFACE
    HAVE_CONFIG_H=1
    USE_LOCALE=1
    POWWOW_DIR="${CMAKE_INSTALL_PREFIX}/share/powwow"
    PLUGIN_DIR="${CMAKE_INSTALL_PREFIX}/lib/powwow"
    VERSION="${PROJECT_VERSION}"
)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_definitions(powwow_platform_definitions INTERFACE _XOPEN_SOURCE=700)
elseif(APPLE)
    target_compile_definitions(powwow_platform_definitions INTERFACE _DARWIN_C_SOURCE=1 USE_BSD_LIKE=1)
endif()

if(UNIX)
    target_compile_definitions(powwow_platform_definitions INTERFACE
        HAVE_GETOPT_LONG=1
        USE_REGEXP=1
        USE_SHELL=1
        HAVE_STDIO_H=1
        HAVE_STDLIB_H=1
        HAVE_STRING_H=1
        HAVE_UNISTD_H=1
        HAVE_SYS_STAT_H=1
        HAVE_SYS_TYPES_H=1
    )
endif()

# Modern GnuTLS dependency discovery via pkg-config
if(ENABLE_TLS AND UNIX)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(GNUTLS IMPORTED_TARGET gnutls)
    endif()
endif()

enable_testing()

# Component Subdirectories
add_subdirectory(libpowwow)
add_subdirectory(tui)
add_subdirectory(gui)

# CPack Configuration
set(CPACK_PACKAGE_NAME "powtty")
set(CPACK_PACKAGE_VENDOR "PowTTY MUD Client")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "PowTTY MUD client and Qt launcher")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/MUME/PowTTY")

if(WIN32)
    set(CPACK_GENERATOR "ZIP;NSIS")
elseif(APPLE)
    set(CPACK_GENERATOR "DragNDrop;TGZ")
else()
    set(CPACK_GENERATOR "TGZ;DEB")
endif()

include(CPack)

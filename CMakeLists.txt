cmake_minimum_required(VERSION 3.16)
project(powwow VERSION 2.0.0 LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Create dummy config.h in binary dir for CMake builds
if(NOT EXISTS "${CMAKE_BINARY_DIR}/config.h")
    file(WRITE "${CMAKE_BINARY_DIR}/config.h" "/* Generated by CMake */\n")
endif()

if(APPLE)
    include_directories(/opt/homebrew/include /usr/local/include)
    link_directories(/opt/homebrew/lib /usr/local/lib)
endif()

if(WIN32 AND MINGW)
    set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -static -static-libgcc -static-libstdc++")
endif()

include_directories("${CMAKE_BINARY_DIR}" "${CMAKE_SOURCE_DIR}" "${CMAKE_SOURCE_DIR}/src")

# Options
option(ENABLE_TLS "Enable TLS via GnuTLS" ON)
option(BUILD_GUI "Build Qt6 GUI Launcher and Terminal" ON)

# Definitions and config header simulation for C core
add_definitions(-DHAVE_CONFIG_H=1)
add_definitions(-DUSE_LOCALE=1)
add_definitions(-DPOWWOW_DIR="${CMAKE_INSTALL_PREFIX}/share/powwow")
add_definitions(-DPLUGIN_DIR="${CMAKE_INSTALL_PREFIX}/lib/powwow")
add_definitions(-DVERSION="${PROJECT_VERSION}")

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_definitions(-D_XOPEN_SOURCE=700)
elseif(APPLE)
    add_definitions(-D_DARWIN_C_SOURCE=1)
    add_definitions(-DUSE_BSD_LIKE=1)
endif()

if(UNIX)
    add_definitions(-DHAVE_GETOPT_LONG=1)
    add_definitions(-DUSE_REGEXP=1)
    add_definitions(-DUSE_SHELL=1)
    add_definitions(-DHAVE_STDIO_H=1)
    add_definitions(-DHAVE_STDLIB_H=1)
    add_definitions(-DHAVE_STRING_H=1)
    add_definitions(-DHAVE_UNISTD_H=1)
    add_definitions(-DHAVE_SYS_STAT_H=1)
    add_definitions(-DHAVE_SYS_TYPES_H=1)
endif()

if(ENABLE_TLS AND UNIX)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(GNUTLS gnutls)
    endif()
    if(NOT GNUTLS_FOUND)
        find_library(GNUTLS_LIB gnutls PATHS /opt/homebrew/lib /usr/local/lib)
        find_path(GNUTLS_INCLUDE_DIR gnutls/gnutls.h PATHS /opt/homebrew/include /usr/local/include)
        if(GNUTLS_LIB AND GNUTLS_INCLUDE_DIR)
            set(GNUTLS_FOUND TRUE)
            set(GNUTLS_LIBRARIES ${GNUTLS_LIB})
            set(GNUTLS_INCLUDE_DIRS ${GNUTLS_INCLUDE_DIR})
        endif()
    endif()

    if(GNUTLS_FOUND)
        add_definitions(-DUSE_TLS=1 -DHAVE_GNUTLS=1)
    endif()
endif()

# C source files for POSIX powwow CLI (Linux/macOS)
if(NOT WIN32)
    find_package(Curses QUIET)
    set(POWWOW_C_SOURCES
        src/beam.c
        src/cmd.c
        src/log.c
        src/edit.c
        src/cmd2.c
        src/eval.c
        src/utils.c
        src/main.c
        src/tcp.c
        src/list.c
        src/map.c
        src/tty.c
        src/ptr.c
        src/tls.c
        src/charset.c
        src/width.c
        src/sgr.c
    )

    add_executable(powwow ${POWWOW_C_SOURCES})

    if(CURSES_FOUND)
        target_include_directories(powwow PRIVATE ${CURSES_INCLUDE_DIRS})
        target_link_libraries(powwow PRIVATE ${CURSES_LIBRARIES})
    else()
        target_link_libraries(powwow PRIVATE ncurses)
    endif()

    if(GNUTLS_FOUND)
        target_include_directories(powwow PRIVATE ${GNUTLS_INCLUDE_DIRS})
        target_link_libraries(powwow PRIVATE ${GNUTLS_LIBRARIES})
    endif()

    if(CMAKE_DL_LIBS)
        target_link_libraries(powwow PRIVATE ${CMAKE_DL_LIBS})
        add_definitions(-DHAVE_LIBDL=1)
    endif()

    # powwow utilities
    add_executable(powwow-muc src/powwow-muc.c)
    add_executable(powwow-movieplay src/powwow-movieplay.c)

    if(CURSES_FOUND)
        target_include_directories(powwow-muc PRIVATE ${CURSES_INCLUDE_DIRS})
        target_link_libraries(powwow-muc PRIVATE ${CURSES_LIBRARIES})
        target_include_directories(powwow-movieplay PRIVATE ${CURSES_INCLUDE_DIRS})
        target_link_libraries(powwow-movieplay PRIVATE ${CURSES_LIBRARIES})
    else()
        target_link_libraries(powwow-muc PRIVATE ncurses)
        target_link_libraries(powwow-movieplay PRIVATE ncurses)
    endif()

    # Tests
    enable_testing()
    add_executable(test_charset src/test_charset.c src/charset.c src/width.c)
    add_test(NAME test_charset COMMAND test_charset)

    add_executable(test_sgr src/test_sgr.c src/sgr.c)
    add_test(NAME test_sgr COMMAND test_sgr)

    install(TARGETS powwow powwow-muc powwow-movieplay DESTINATION bin)
endif()

# Qt6 GUI Application (PowTTY)
if(BUILD_GUI)
    find_package(Qt6 COMPONENTS Core Widgets Network QUIET)
    if(Qt6_FOUND)
        set(CMAKE_AUTOMOC ON)
        set(CMAKE_AUTORCC ON)
        set(CMAKE_AUTOUIC ON)

        # MSVC Static Runtime configuration if static build
        if(MSVC AND NOT BUILD_SHARED_LIBS)
            set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        endif()

        set(GUI_SOURCES
            gui/main.cpp
            gui/mainwindow.cpp
            gui/mainwindow.h
            gui/terminalwidget.cpp
            gui/terminalwidget.h
            gui/profilemanager.cpp
            gui/profilemanager.h
            gui/resources.qrc
        )

        if(WIN32)
            list(APPEND GUI_SOURCES gui/powtty.rc)
        endif()

        if(APPLE)
            set(MACOS_ICON_FILE "${CMAKE_CURRENT_SOURCE_DIR}/gui/resources/icons/icon.icns")
            set_source_files_properties(${MACOS_ICON_FILE} PROPERTIES MACOSX_PACKAGE_LOCATION "Resources")
            add_executable(powtty MACOSX_BUNDLE ${GUI_SOURCES} ${MACOS_ICON_FILE})
            set_target_properties(powtty PROPERTIES
                MACOSX_BUNDLE_BUNDLE_NAME "PowTTY"
                MACOSX_BUNDLE_GUI_IDENTIFIER "org.mume.powtty"
                MACOSX_BUNDLE_ICON_FILE "icon.icns"
            )
        else()
            add_executable(powtty ${GUI_SOURCES})
        endif()

        target_link_libraries(powtty PRIVATE
            Qt6::Core
            Qt6::Widgets
            Qt6::Network
        )

        install(TARGETS powtty DESTINATION bin)

        # Deployment step using macdeployqt / windeployqt in CMake install/post-build
        if(WIN32)
            find_program(WINDEPLOYQT_EXECUTABLE windeployqt HINTS ${Qt6_DIR}/../../../bin)
            if(WINDEPLOYQT_EXECUTABLE)
                add_custom_command(TARGET powtty POST_BUILD
                    COMMAND ${WINDEPLOYQT_EXECUTABLE} --no-compiler-runtime $<TARGET_FILE:powtty>
                    COMMENT "Running windeployqt..."
                )
            endif()
        elseif(APPLE)
            find_program(MACDEPLOYQT_EXECUTABLE macdeployqt HINTS ${Qt6_DIR}/../../../bin)
            if(MACDEPLOYQT_EXECUTABLE)
                add_custom_command(TARGET powtty POST_BUILD
                    COMMAND ${MACDEPLOYQT_EXECUTABLE} $<TARGET_BUNDLE_DIR:powtty> -always-overwrite
                    COMMENT "Running macdeployqt..."
                )
            endif()
        endif()
    else()
        message(STATUS "Qt6 not found. Skipping powtty build.")
    endif()
endif()

# CPack Configuration
set(CPACK_PACKAGE_NAME "powtty")
set(CPACK_PACKAGE_VENDOR "PowTTY MUD Client")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "PowTTY MUD client and Qt launcher")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/MUME/PowTTY")

if(WIN32)
    set(CPACK_GENERATOR "ZIP;NSIS")
elseif(APPLE)
    set(CPACK_GENERATOR "DragNDrop;TGZ")
else()
    set(CPACK_GENERATOR "TGZ;DEB")
endif()

include(CPack)
